const isProd = process.env.NODE_ENV === "production";

/**
 * NOTE ON SECURITY HEADERS
 * `X-Frame-Options` / `frame-ancestors` are intentionally NOT set here: preview and
 * staging environments embed the app in an iframe. Set framing rules at the hosting
 * layer once the production domain is confirmed.
 */
/**
 * A deliberately conservative CSP: it locks down script/style/object/base origins
 * without needing nonces (Next.js streams inline bootstrap scripts). Framing rules
 * are still delegated to the hosting layer. If GA4 / Meta Pixel are added later,
 * extend `script-src` and `connect-src` with their domains in the same commit.
 */
const csp = [
  "default-src 'self'",
  "base-uri 'self'",
  "object-src 'none'",
  "img-src 'self' data: blob:",
  "font-src 'self'",
  "style-src 'self' 'unsafe-inline'",
  "script-src 'self' 'unsafe-inline'",
  "connect-src 'self'",
  "form-action 'self'",
].join("; ");

const securityHeaders = [
  { key: "Content-Security-Policy", value: csp },
  { key: "X-Content-Type-Options", value: "nosniff" },
  { key: "Referrer-Policy", value: "strict-origin-when-cross-origin" },
  { key: "Permissions-Policy", value: "camera=(), microphone=(), geolocation=(self)" },
  { key: "X-DNS-Prefetch-Control", value: "on" },
  ...(isProd
    ? [{ key: "Strict-Transport-Security", value: "max-age=63072000; includeSubDomains; preload" }]
    : []),
];
