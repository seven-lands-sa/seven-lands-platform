/**
 * ⚠️ TEMPORARY — PHASE A SPIKE SCRIPT. REMOVE WITH src/spike/ AFTER THE SPIKE.
 *
 * Proves, against the real development PostgreSQL + Local API:
 *   1. first-user creation (auth collection works)
 *   2. Create / Read / Update / Delete
 *   3. access control is enforced when NOT overriding access
 *   4. media upload + sharp derivative generation
 *
 * Run: npx payload run src/spike/phase-a-check.ts
 */
import fs from 'fs'
import os from 'os'
import path from 'path'

import { getPayload } from 'payload'
import sharp from 'sharp'

import config from '@payload-config'

const payload = await getPayload({ config })

let failures = 0
const check = (name: string, ok: boolean, detail = '') => {
  if (!ok) failures += 1
  console.log(`${ok ? 'PASS' : 'FAIL'}  ${name}${detail ? ` — ${detail}` : ''}`)
}

// ── 1. first user ────────────────────────────────────────────────────────────
const userCount = await payload.count({ collection: 'spike-users' })
if (userCount.totalDocs === 0) {
  const created = await payload.create({
    collection: 'spike-users',
    data: {
      email: 'spike-admin@seven-lands.local',
      password: 'PhaseA-Spike-2026!',
      name: 'Phase A Spike Admin',
    },
  })
  check('first user created', Boolean(created.id), `id=${created.id}`)
} else {
  check('first user exists', true, `count=${userCount.totalDocs}`)
}
const users = await payload.find({ collection: 'spike-users', limit: 1 })
console.log(`      admin login: ${users.docs[0]?.email ?? 'NONE'}`)

// ── 2. CRUD ─────────────────────────────────────────────────────────────────
const created = await payload.create({
  collection: 'spike-items',
  data: { title: 'Phase A CRUD probe', note: 'created by phase-a-check.ts' },
})
check('CREATE', Boolean(created.id), `id=${created.id}`)

const read = await payload.findByID({ collection: 'spike-items', id: created.id })
check('READ', read.title === 'Phase A CRUD probe', `title="${read.title}"`)

const updated = await payload.update({
  collection: 'spike-items',
  id: created.id,
  data: { title: 'Phase A CRUD probe (updated)' },
})
check('UPDATE', updated.title === 'Phase A CRUD probe (updated)', `title="${updated.title}"`)

const afterUpdate = await payload.findByID({ collection: 'spike-items', id: created.id })
check('READ-AFTER-UPDATE', afterUpdate.title === 'Phase A CRUD probe (updated)')

await payload.delete({ collection: 'spike-items', id: created.id })
let deleted = false
try {
  await payload.findByID({ collection: 'spike-items', id: created.id })
} catch {
  deleted = true
}
check('DELETE', deleted, 'subsequent read throws NotFound')

// ── 3. access control is real (no overrideAccess) ───────────────────────────
let denied = false
try {
  await payload.find({ collection: 'spike-items', overrideAccess: false })
} catch (err) {
  denied = err instanceof Error && /forbidden|unauthorized|not allowed/i.test(err.message)
  if (!denied) console.log(`      unexpected error: ${(err as Error).message}`)
}
check('access control denies anonymous reads', denied)

// ── 4. media upload + image processing ──────────────────────────────────────
const srcPath = path.join(os.tmpdir(), 'phase-a-upload.png')
await sharp({
  create: { width: 1200, height: 800, channels: 3, background: { r: 122, g: 156, b: 198 } },
})
  .png()
  .toFile(srcPath)
check('test image generated', fs.existsSync(srcPath), `${fs.statSync(srcPath).size} bytes`)

const media = await payload.create({
  collection: 'spike-media',
  data: { alt: 'صورة اختبار المرحلة أ' },
  filePath: srcPath,
})
const m = media as unknown as {
  filename: string
  width?: number
  height?: number
  sizes?: Record<string, { width?: number; height?: number; filename?: string }>
}
check('MEDIA UPLOAD', Boolean(m.filename), `filename=${m.filename} alt="${media.alt}"`)
check(
  'IMAGE PROCESSING — dimensions detected',
  m.width === 1200 && m.height === 800,
  `${m.width}x${m.height}`,
)
check(
  'IMAGE PROCESSING — thumbnail derivative (400px)',
  m.sizes?.thumbnail?.width === 400,
  `thumbnail=${m.sizes?.thumbnail?.width}x${m.sizes?.thumbnail?.height} (${m.sizes?.thumbnail?.filename})`,
)
check(
  'IMAGE PROCESSING — card derivative (768px)',
  m.sizes?.card?.width === 768,
  `card=${m.sizes?.card?.width}x${m.sizes?.card?.height} (${m.sizes?.card?.filename})`,
)
const stockDir = path.resolve(process.cwd(), 'media-spike')
const files = fs.existsSync(stockDir) ? fs.readdirSync(stockDir) : []
check('STORAGE — files on local disk adapter', files.length >= 3, files.join(', '))

console.log(`\n${failures === 0 ? 'ALL CHECKS PASSED' : `${failures} CHECK(S) FAILED`}`)
process.exit(failures === 0 ? 0 : 1)
